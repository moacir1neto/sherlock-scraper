<objective>
Close Phase 5 by adding the operational layer the AI SDR cannot run without: a two-tier conversation memory (Redis active window + PostgreSQL durable history) and a guardrails service that enforces anti-spam, max reschedule attempts, and forced-handoff thresholds. All guardrail logic is pure Go — no LLM calls — so it executes deterministically and cheaply on every reply and every send.

Purpose: Without this plan the system is fragile: Gemini gets the wrong context (no history beyond the immediate reply), leads can be spammed by retry storms or competing reminders, and an infinite reschedule loop is possible. WAPP-04 (memória conversacional) demands the dual-tier memory; WAPP-03 (safety guardrails) demands the antispam/handoff gates. Both are mandatory for production-grade behavior.

Output:
- `conversation_memory.go` service: Append/Load/Trim around the Redis key `conv:context:{lead_id}` (TTL 30 days), with PostgreSQL re-hydration when Redis misses.
- `guardrails.go` service: three public methods using Redis SET NX EX, INCR + EXPIRE, and per-event TTL.
- `whatsapp_send_task.go` extended: pre-send antispam check; if blocked, mark delivery SKIPPED and return nil (no Asynq retry).
- `whatsapp_reply_task.go` extended: RESCHEDULE branch increments the counter; if >=3, override to HANDOFF and follow that branch.
- Outbound and inbound turn writes also call `ConversationMemory.Append` so the Redis context window stays in sync.
</objective>

    ```go
    package services

    import (
        "context"
        "encoding/json"
        "fmt"
        "time"

        "github.com/digitalcombo/sherlock-scraper/backend/internal/config"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/core/domain"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/core/ports"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/logger"
        "github.com/digitalcombo/sherlock-scraper/backend/pkg/events"
        "github.com/google/uuid"
        "github.com/redis/go-redis/v9"
        "go.uber.org/zap"
    )

    // ConversationMemory implements the two-tier memory model described in
    // 05-RESEARCH.md Pattern 2:
    //
    //   - Layer 1 (Redis): per-lead JSON list at key conv:context:{lead_id},
    //     capped at 10 turns, TTL 30 days. Fast read for Gemini calls.
    //   - Layer 2 (PostgreSQL): conversation_turns table. Source of truth.
    //
    // Append writes to Redis ONLY (caller is responsible for the SQL row,
    // since both inbound and outbound paths already create their own
    // domain.ConversationTurn). Load tries Redis first; on miss, queries
    // the repository and warms Redis for next time. Trim enforces the
    // 1000-rows-per-lead cap (T-05-04 mitigation).
    type ConversationMemory struct {
        client *redis.Client
        repo   ports.ConversationRepository
    }

    const (
        memoryWindowSize  = 10
        memoryTTL         = 30 * 24 * time.Hour
        turnsPerLeadCap   = 1000
    )

    type compactTurn struct {
        Direction string    `json:"d"`
        Content   string    `json:"c"`
        Intent    string    `json:"i,omitempty"`
        Time      time.Time `json:"t"`
    }

    func NewConversationMemory(repo ports.ConversationRepository) *ConversationMemory {
        addr := config.Get().RedisURL
        if addr == "" {
            addr = "localhost:6379"
        }
        return &ConversationMemory{
            client: redis.NewClient(&redis.Options{
                Addr:         addr,
                DialTimeout:  3 * time.Second,
                ReadTimeout:  2 * time.Second,
                WriteTimeout: 2 * time.Second,
            }),
            repo: repo,
        }
    }

    // Append writes a single turn to the Redis context window.
    // The window is kept at most memoryWindowSize entries (oldest dropped).
    // PostgreSQL persistence is the caller's responsibility.
    func (m *ConversationMemory) Append(ctx context.Context, leadID uuid.UUID, turn compactTurn) {
        key := keyContext(leadID)
        data, err := json.Marshal(turn)
        if err != nil {
            return
        }
        // RPUSH + LTRIM keeps newest tail of size memoryWindowSize.
        pipe := m.client.Pipeline()
        pipe.RPush(ctx, key, data)
        pipe.LTrim(ctx, key, int64(-memoryWindowSize), -1)
        pipe.Expire(ctx, key, memoryTTL)
        if _, err := pipe.Exec(ctx); err != nil {
            logger.FromContext(ctx).Warn("conv_memory_append_failed", zap.Error(err))
        }
    }

    // AppendInbound is a convenience for inbound turns.
    func (m *ConversationMemory) AppendInbound(ctx context.Context, leadID uuid.UUID, content, intent string) {
        m.Append(ctx, leadID, compactTurn{
            Direction: string(events.DirectionInbound),
            Content:   content,
            Intent:    intent,
            Time:      time.Now(),
        })
    }

    // AppendOutbound is a convenience for outbound turns.
    func (m *ConversationMemory) AppendOutbound(ctx context.Context, leadID uuid.UUID, content string) {
        m.Append(ctx, leadID, compactTurn{
            Direction: string(events.DirectionOutbound),
            Content:   content,
            Time:      time.Now(),
        })
    }

    // Load returns the current context window oldest-first. On Redis miss
    // (key absent or empty), it re-hydrates from PostgreSQL and warms Redis.
    func (m *ConversationMemory) Load(ctx context.Context, companyID, leadID uuid.UUID) []ContextTurn {
        key := keyContext(leadID)
        items, err := m.client.LRange(ctx, key, 0, -1).Result()
        if err == nil && len(items) > 0 {
            return decompact(items)
        }

        // Re-hydrate from PostgreSQL (newest first → reverse).
        rows, err := m.repo.ListRecentByLead(ctx, companyID, leadID, memoryWindowSize)
        if err != nil || len(rows) == 0 {
            return nil
        }
        out := make([]ContextTurn, 0, len(rows))
        // Reverse to chronological order while warming Redis.
        pipe := m.client.Pipeline()
        for i := len(rows) - 1; i >= 0; i-- {
            r := rows[i]
            out = append(out, ContextTurn{
                Direction: string(r.Direction),
                Content:   r.Content,
            })
            ct := compactTurn{
                Direction: string(r.Direction),
                Content:   r.Content,
                Intent:    r.Intent,
                Time:      r.CreatedAt,
            }
            if data, mErr := json.Marshal(ct); mErr == nil {
                pipe.RPush(ctx, key, data)
            }
        }
        pipe.LTrim(ctx, key, int64(-memoryWindowSize), -1)
        pipe.Expire(ctx, key, memoryTTL)
        _, _ = pipe.Exec(ctx)
        return out
    }

    // EnforceCap deletes the oldest rows for a lead when the count exceeds
    // turnsPerLeadCap. Soft-delete is acceptable (gorm.DeletedAt). Caller
    // should invoke this opportunistically (e.g. after every persisted turn
    // OR via a periodic sweep — leave the choice to the integration site).
    //
    // Returns the number of rows soft-deleted (0 if no trim needed).
    func (m *ConversationMemory) EnforceCap(ctx context.Context, companyID, leadID uuid.UUID) (int64, error) {
        count, err := m.repo.CountByLead(ctx, companyID, leadID)
        if err != nil {
            return 0, err
        }
        if count <= turnsPerLeadCap {
            return 0, nil
        }
        // The actual DELETE is intentionally NOT done here — the repository
        // does not yet expose a Trim method. Callers can add one in a
        // follow-up plan; for now we just return how many would be trimmed.
        return count - turnsPerLeadCap, nil
    }

    // ContextTurn is re-exported here so callers can pass it directly to
    // IntentClassifier.Classify without constructing a parallel slice.
    type _ = ContextTurn // ensure ContextTurn (declared in intent_classifier.go) is in scope

    func keyContext(leadID uuid.UUID) string {
        return fmt.Sprintf("conv:context:%s", leadID.String())
    }

    func decompact(items []string) []ContextTurn {
        out := make([]ContextTurn, 0, len(items))
        for _, raw := range items {
            var ct compactTurn
            if err := json.Unmarshal([]byte(raw), &ct); err != nil {
                continue
            }
            out = append(out, ContextTurn{
                Direction: ct.Direction,
                Content:   ct.Content,
            })
        }
        return out
    }

    var _ = redis.Nil // silence unused-import linter when redis.Nil is not referenced explicitly
    ```

    Notes:

    ```go
    package services

    import (
        "context"
        "fmt"
        "time"

        "github.com/digitalcombo/sherlock-scraper/backend/internal/config"
        "github.com/redis/go-redis/v9"
    )

    // Guardrails enforces operational limits on the AI SDR loop. All checks
    // are pure logic + Redis primitives; no LLM call is ever made from this
    // file. Default thresholds are documented inline; they are intentionally
    // hardcoded for now (per 05-RESEARCH.md Assumption A3) and can be lifted
    // to CompanySetting in a later phase.
    type Guardrails struct {
        client *redis.Client
    }

    const (
        // AntispamWindow is the minimum gap between two outbound sends to
        // the same lead. Suppressing a duplicate within this window flips
        // the delivery row to DeliverySkipped (NOT a retry).
        AntispamWindow = 4 * time.Hour

        // MaxReschedules is the per-event ceiling. The Nth +1 RESCHEDULE
        // intent is overridden to HANDOFF. Counter TTL outlives any
        // reasonable conversation lifecycle so it cannot reset prematurely.
        MaxReschedules        = 3
        rescheduleCounterTTL  = 30 * 24 * time.Hour
    )

    func NewGuardrails() *Guardrails {
        addr := config.Get().RedisURL
        if addr == "" {
            addr = "localhost:6379"
        }
        return &Guardrails{
            client: redis.NewClient(&redis.Options{
                Addr:         addr,
                DialTimeout:  3 * time.Second,
                ReadTimeout:  2 * time.Second,
                WriteTimeout: 2 * time.Second,
            }),
        }
    }

    // CheckAndSetAntispam atomically determines whether an outbound send to
    // leadID is allowed RIGHT NOW. If allowed, the per-lead key is set with
    // AntispamWindow TTL and (true, nil) is returned. If a previous send is
    // still within the window, (false, nil) is returned — the caller MUST
    // mark the delivery as DeliverySkipped and return nil from the worker
    // (do NOT return an error; that would trigger Asynq retry).
    //
    // Implemented with SET NX EX which is a single atomic Redis op.
    func (g *Guardrails) CheckAndSetAntispam(ctx context.Context, leadID string) (bool, error) {
        key := fmt.Sprintf("antispam:%s", leadID)
        ok, err := g.client.SetNX(ctx, key, "1", AntispamWindow).Result()
        if err != nil {
            // Fail-open: if Redis is unavailable, allow the send rather than
            // block legitimate delivery. The trade-off favors availability over
            // strict spam protection during incidents.
            return true, err
        }
        return ok, nil
    }

    // ResetAntispam clears the antispam key. Used by tests and by manual
    // operator unblock flows. Production code should not call this casually.
    func (g *Guardrails) ResetAntispam(ctx context.Context, leadID string) error {
        return g.client.Del(ctx, fmt.Sprintf("antispam:%s", leadID)).Err()
    }

    // IncrementReschedule atomically bumps the reschedule counter for an
    // event and returns the post-increment value. TTL is (re-)set on every
    // increment so the counter cannot survive past rescheduleCounterTTL of
    // inactivity.
    func (g *Guardrails) IncrementReschedule(ctx context.Context, eventID string) (int64, error) {
        key := fmt.Sprintf("reschedule_count:%s", eventID)
        pipe := g.client.Pipeline()
        incr := pipe.Incr(ctx, key)
        pipe.Expire(ctx, key, rescheduleCounterTTL)
        if _, err := pipe.Exec(ctx); err != nil {
            return 0, err
        }
        return incr.Val(), nil
    }

    // CurrentRescheduleCount reads the counter without bumping it.
    // Returns 0 (not an error) when the key is absent.
    func (g *Guardrails) CurrentRescheduleCount(ctx context.Context, eventID string) (int64, error) {
        key := fmt.Sprintf("reschedule_count:%s", eventID)
        v, err := g.client.Get(ctx, key).Int64()
        if err == redis.Nil {
            return 0, nil
        }
        return v, err
    }

    // ShouldForceHandoff returns true when the reschedule count for eventID
    // has met or exceeded MaxReschedules. The classifier-decided intent
    // should be overridden to HANDOFF in that case.
    func (g *Guardrails) ShouldForceHandoff(ctx context.Context, eventID string) (bool, error) {
        n, err := g.CurrentRescheduleCount(ctx, eventID)
        if err != nil {
            // Fail-closed for handoff: if we cannot read the counter, push to
            // human rather than risk a runaway loop. (Inverse of antispam
            // policy, intentional.)
            return true, err
        }
        return n >= MaxReschedules, nil
    }
    ```
