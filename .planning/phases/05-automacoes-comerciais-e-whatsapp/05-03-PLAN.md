<objective>
Deliver the AI-driven inbound half of the SDR loop: a hybrid template service (WAPP-01), an intent classifier built around `events.ParseIntent` (WAPP-03), and a reply pipeline (WAPP-02) wired through a new InternalAuth-protected webhook so WhatsMiau can forward incoming WhatsApp messages to the backend without timing out.

Purpose: Closes the "alerts but no engagement" gap from Phase 4 by transforming inbound replies into structured intents and routing each intent to the appropriate next action (no-op confirm, rule-based reschedule, AI-empathetic objection, or handoff). The static template fallback guarantees delivery even when Gemini is unavailable (Pattern 3 in 05-RESEARCH.md). All Gemini calls go through `pkg/llm.DefaultLLMModel` — no occurrence of `gemini-2.0-flash` in any new file.

Output:
- `template_service.go` with two render functions; both fall back to a deterministic static string when Gemini fails.
- `intent_classifier.go` with a single `Classify(ctx, reply, history)` method that emits a low-token Gemini prompt and pipes the response through `events.ParseIntent` (which forces UNKNOWN on any unrecognized output).
- `whatsapp_reply_task.go` with payload struct, factory, handler, and the five intent branches.
- `whatsapp_webhook.go` HTTP handler that validates the body, enqueues `TaskWhatsAppReply`, and returns 202 immediately.
- `server.go` registers the new task type. `main.go` registers the webhook route under the existing `internal := api.Group("/internal", middlewares.InternalAuth())` group.
</objective>

    ```go
    package services

    import (
        "bytes"
        "context"
        "encoding/json"
        "fmt"
        "io"
        "net/http"
        "strings"
        "time"

        "github.com/digitalcombo/sherlock-scraper/backend/internal/config"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/logger"
        "github.com/digitalcombo/sherlock-scraper/backend/pkg/llm"
        "go.uber.org/zap"
    )

    // TemplateService renders WhatsApp message bodies using a hybrid strategy:
    // a deterministic static base + an optional Gemini "tone-fill" pass.
    // If the LLM call fails or produces an empty/oversize response, the static
    // base is returned. Delivery is NEVER blocked on LLM availability.
    //
    // The two public methods cover the only template categories needed in
    // Phase 5; new categories should be added here as new methods, not as
    // free-form prompts elsewhere.
    type TemplateService struct {
        httpClient *http.Client
    }

    func NewTemplateService() *TemplateService {
        return &TemplateService{
            httpClient: &http.Client{Timeout: 8 * time.Second},
        }
    }

    // RenderMeetingReminder produces the body of a meeting-reminder message.
    // baseVars carries: lead_name, scheduled_at, label (24h/1h/15m), tone (optional).
    func (s *TemplateService) RenderMeetingReminder(ctx context.Context, baseVars map[string]string) string {
        leadName := baseVars["lead_name"]
        scheduledAt := baseVars["scheduled_at"]
        label := baseVars["label"]
        tone := strings.TrimSpace(baseVars["tone"])

        base := fmt.Sprintf(
            "Olá %s! Lembrete (%s): nossa reunião está marcada para %s. Posso confirmar?",
            leadName, label, scheduledAt,
        )

        if tone == "" {
            return base
        }

        prompt := fmt.Sprintf(
            "Reescreva a mensagem abaixo mantendo o conteúdo idêntico, "+
                "apenas ajustando o tom para %q. Devolva APENAS o texto final, "+
                "sem aspas, sem prefixos, máximo 320 caracteres.\n\nMensagem:\n%s",
            tone, base,
        )
        out := s.callGeminiSafe(ctx, prompt)
        if out == "" {
            return base
        }
        return out
    }

    // RenderObjectionResponse produces a brief empathetic reply to a lead
    // objection. It uses the static fallback whenever Gemini fails.
    func (s *TemplateService) RenderObjectionResponse(ctx context.Context, leadReply, leadName string) string {
        base := fmt.Sprintf(
            "Entendo, %s. Sem pressão — quando fizer sentido para você, é só me avisar e seguimos.",
            leadName,
        )
        prompt := fmt.Sprintf(
            "Você é um SDR. O lead %q acabou de responder com uma objeção: %q. "+
                "Escreva uma resposta curta (até 280 caracteres), empática e sem fechar a porta. "+
                "Devolva APENAS o texto da resposta.",
            leadName, leadReply,
        )
        out := s.callGeminiSafe(ctx, prompt)
        if out == "" {
            return base
        }
        return out
    }

    // callGeminiSafe wraps the Gemini REST call with timeout, length guards,
    // and graceful error handling. Returns "" on any failure.
    func (s *TemplateService) callGeminiSafe(ctx context.Context, prompt string) string {
        l := logger.FromContext(ctx)
        apiKey := config.Get().GeminiAPIKey
        if apiKey == "" {
            return ""
        }
        reqBody := map[string]interface{}{
            "contents": []map[string]interface{}{
                {"parts": []map[string]interface{}{{"text": prompt}}},
            },
        }
        body, err := json.Marshal(reqBody)
        if err != nil {
            return ""
        }
        req, err := http.NewRequestWithContext(ctx, "POST",
            llm.GeminiEndpointURL(apiKey),
            bytes.NewBuffer(body),
        )
        if err != nil {
            return ""
        }
        req.Header.Set("Content-Type", "application/json")
        resp, err := s.httpClient.Do(req)
        if err != nil {
            l.Warn("template_gemini_request_failed", zap.Error(err))
            return ""
        }
        defer resp.Body.Close()
        if resp.StatusCode >= 400 {
            return ""
        }
        var gem struct {
            Candidates []struct {
                Content struct {
                    Parts []struct {
                        Text string `json:"text"`
                    } `json:"parts"`
                } `json:"content"`
            } `json:"candidates"`
        }
        raw, _ := io.ReadAll(io.LimitReader(resp.Body, 8192))
        if err := json.Unmarshal(raw, &gem); err != nil {
            return ""
        }
        if len(gem.Candidates) == 0 || len(gem.Candidates[0].Content.Parts) == 0 {
            return ""
        }
        out := strings.TrimSpace(gem.Candidates[0].Content.Parts[0].Text)
        // Length guard against runaway output.
        if len(out) == 0 || len(out) > 1000 {
            return ""
        }
        return out
    }
    ```

    ```go
    package services

    import (
        "bytes"
        "context"
        "encoding/json"
        "fmt"
        "io"
        "net/http"
        "strings"
        "time"

        "github.com/digitalcombo/sherlock-scraper/backend/internal/config"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/logger"
        "github.com/digitalcombo/sherlock-scraper/backend/pkg/events"
        "github.com/digitalcombo/sherlock-scraper/backend/pkg/llm"
        "go.uber.org/zap"
    )

    // IntentClassifier wraps the Gemini call that maps a lead reply into one of
    // the closed-set IntentType values. Any failure (network, parse, unknown
    // word) yields IntentUnknown — the caller is expected to treat UNKNOWN as
    // a no-op + log signal.
    type IntentClassifier struct {
        httpClient *http.Client
    }

    func NewIntentClassifier() *IntentClassifier {
        return &IntentClassifier{httpClient: &http.Client{Timeout: 6 * time.Second}}
    }

    // ContextTurn is a compact view of a prior conversation turn passed to the
    // classifier prompt. The caller produces it from the Redis context window
    // or the conversation_turns table.
    type ContextTurn struct {
        Direction string // "INBOUND" or "OUTBOUND"
        Content   string
    }

    // Classify returns one of the five IntentType constants. Never returns
    // an error: any failure path collapses to IntentUnknown so callers can
    // make a deterministic guardrail decision.
    func (c *IntentClassifier) Classify(ctx context.Context, replyText string, history []ContextTurn) events.IntentType {
        l := logger.FromContext(ctx)
        apiKey := config.Get().GeminiAPIKey
        if apiKey == "" {
            return events.IntentUnknown
        }

        var sb strings.Builder
        sb.WriteString("Classifique a resposta abaixo em UMA palavra exata: CONFIRM, RESCHEDULE, OBJECTION, HANDOFF ou UNKNOWN. ")
        sb.WriteString("Responda apenas com a palavra, sem texto adicional, sem pontuação.\n\n")
        if len(history) > 0 {
            sb.WriteString("Histórico recente (mais antigo primeiro):\n")
            for _, t := range history {
                fmt.Fprintf(&sb, "- [%s] %s\n", t.Direction, truncate(t.Content, 240))
            }
            sb.WriteString("\n")
        }
        fmt.Fprintf(&sb, "Resposta do lead:\n<<<%s>>>\n", truncate(replyText, 800))

        reqBody := map[string]interface{}{
            "contents": []map[string]interface{}{
                {"parts": []map[string]interface{}{{"text": sb.String()}}},
            },
        }
        body, err := json.Marshal(reqBody)
        if err != nil {
            return events.IntentUnknown
        }
        req, err := http.NewRequestWithContext(ctx, "POST",
            llm.GeminiEndpointURL(apiKey), bytes.NewBuffer(body))
        if err != nil {
            return events.IntentUnknown
        }
        req.Header.Set("Content-Type", "application/json")
        resp, err := c.httpClient.Do(req)
        if err != nil {
            l.Warn("intent_classifier_request_failed", zap.Error(err))
            return events.IntentUnknown
        }
        defer resp.Body.Close()
        if resp.StatusCode >= 400 {
            return events.IntentUnknown
        }
        var gem struct {
            Candidates []struct {
                Content struct {
                    Parts []struct {
                        Text string `json:"text"`
                    } `json:"parts"`
                } `json:"content"`
            } `json:"candidates"`
        }
        raw, _ := io.ReadAll(io.LimitReader(resp.Body, 4096))
        if err := json.Unmarshal(raw, &gem); err != nil {
            return events.IntentUnknown
        }
        if len(gem.Candidates) == 0 || len(gem.Candidates[0].Content.Parts) == 0 {
            return events.IntentUnknown
        }
        return events.ParseIntent(gem.Candidates[0].Content.Parts[0].Text)
    }

    func truncate(s string, n int) string {
        if len(s) <= n {
            return s
        }
        return s[:n] + "..."
    }
    ```

    ```go
    package queue

    import (
        "context"
        "encoding/json"
        "fmt"
        "time"

        "github.com/digitalcombo/sherlock-scraper/backend/internal/core/domain"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/database"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/logger"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/repositories"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/services"
        "github.com/digitalcombo/sherlock-scraper/backend/internal/sse"
        "github.com/digitalcombo/sherlock-scraper/backend/pkg/events"
        "github.com/google/uuid"
        "github.com/hibiken/asynq"
        "go.uber.org/zap"
    )

    // WhatsAppReplyPayload carries an inbound reply received via the WhatsMiau
    // webhook. CompanyID and InstanceID are required for outbound responses
    // and multi-tenant scoping; BusinessEventID is optional (nil when the
    // reply is unrelated to a scheduled event).
    type WhatsAppReplyPayload struct {
        CompanyID       string `json:"company_id"`
        LeadID          string `json:"lead_id"`
        BusinessEventID string `json:"business_event_id,omitempty"`
        InstanceID      string `json:"instance_id"`
        Phone           string `json:"phone"`
        ReplyText       string `json:"reply_text"`
        TraceID         string `json:"trace_id,omitempty"`
    }

    func NewWhatsAppReplyTask(p WhatsAppReplyPayload) (*asynq.Task, []asynq.Option, error) {
        data, err := json.Marshal(p)
        if err != nil {
            return nil, nil, err
        }
        opts := []asynq.Option{
            asynq.Queue("whatsapp"),
            asynq.MaxRetry(3),
            asynq.Unique(2 * time.Minute),
        }
        return asynq.NewTask(events.TaskWhatsAppReply, data), opts, nil
    }

    // HandleWhatsAppReplyTask is the single entry point for inbound replies.
    // Lifecycle:
    //   1. Persist the inbound turn (pending intent).
    //   2. Load recent context window from PostgreSQL (fallback for Redis miss).
    //   3. Classify intent via Gemini.
    //   4. Update the turn's intent column.
    //   5. Branch on intent: CONFIRM, RESCHEDULE, OBJECTION, HANDOFF, UNKNOWN.
    // Guardrail enforcement (max reschedule, antispam) is layered in Plan 04.
    func HandleWhatsAppReplyTask(ctx context.Context, t *asynq.Task) error {
        var p WhatsAppReplyPayload
        if err := json.Unmarshal(t.Payload(), &p); err != nil {
            return fmt.Errorf("unmarshal reply payload: %w", err)
        }

        l := logger.FromContext(ctx).With(
            zap.String("task", "whatsapp:reply-received"),
            zap.String("lead_id", p.LeadID),
        )

        companyUUID, err := uuid.Parse(p.CompanyID)
        if err != nil {
            return fmt.Errorf("invalid company_id: %w", err)
        }
        leadUUID, err := uuid.Parse(p.LeadID)
        if err != nil {
            return fmt.Errorf("invalid lead_id: %w", err)
        }
        var beUUID *uuid.UUID
        if p.BusinessEventID != "" {
            if u, perr := uuid.Parse(p.BusinessEventID); perr == nil {
                beUUID = &u
            }
        }

        convRepo := repositories.NewConversationRepository(database.DB)
        eventRepo := repositories.NewBusinessEventRepository(database.DB)

        // 1. Persist the inbound turn.
        turn := &domain.ConversationTurn{
            CompanyID: companyUUID,
            LeadID:    leadUUID,
            Direction: events.DirectionInbound,
            Content:   p.ReplyText,
            EventID:   beUUID,
            TraceID:   p.TraceID,
        }
        if err := convRepo.Create(ctx, turn); err != nil {
            l.Error("persist_inbound_turn_failed", zap.Error(err))
        }

        // 2. Build context window (last 10 turns, oldest first).
        history, _ := convRepo.ListRecentByLead(ctx, companyUUID, leadUUID, 10)
        // Reverse for chronological order.
        ctxWindow := make([]services.ContextTurn, 0, len(history))
        for i := len(history) - 1; i >= 0; i-- {
            ctxWindow = append(ctxWindow, services.ContextTurn{
                Direction: string(history[i].Direction),
                Content:   history[i].Content,
            })
        }

        // 3. Classify intent.
        classifier := services.NewIntentClassifier()
        intent := classifier.Classify(ctx, p.ReplyText, ctxWindow)
        l.Info("intent_classified", zap.String("intent", string(intent)))

        // 4. Update the inbound turn with the resolved intent.
        turn.Intent = string(intent)
        if err := database.DB.WithContext(ctx).Save(turn).Error; err != nil {
            l.Warn("turn_intent_update_failed", zap.Error(err))
        }

        // 5. Branch.
        switch intent {
        case events.IntentConfirm:
            if beUUID != nil {
                if be, err := eventRepo.GetByID(ctx, *beUUID); err == nil {
                    be.Status = "CONFIRMED"
                    _ = eventRepo.Update(ctx, be)
                }
            }
            return nil

        case events.IntentReschedule:
            return handleRescheduleIntent(ctx, p, companyUUID, leadUUID, beUUID, eventRepo)

        case events.IntentObjection:
            return handleObjectionIntent(ctx, p, companyUUID, leadUUID, beUUID)

        case events.IntentHandoff:
            return handleHandoffIntent(ctx, p, companyUUID, leadUUID, beUUID, eventRepo)

        case events.IntentUnknown:
            l.Info("intent_unknown_suppressed_outbound")
            return nil
        }
        return nil
    }

    // handleRescheduleIntent: rule-based +1 business-day proposal, no LLM call.
    // Sends a confirmation question to the lead via outbound WhatsApp.
    func handleRescheduleIntent(
        ctx context.Context,
        p WhatsAppReplyPayload,
        companyID, leadID uuid.UUID,
        beUUID *uuid.UUID,
        eventRepo interface {
            GetByID(ctx context.Context, id uuid.UUID) (*domain.BusinessEvent, error)
        },
    ) error {
        proposed := time.Now().Add(24 * time.Hour)
        // Skip Saturday/Sunday.
        for proposed.Weekday() == time.Saturday || proposed.Weekday() == time.Sunday {
            proposed = proposed.Add(24 * time.Hour)
        }
        // Pin to 10:00 local time.
        proposed = time.Date(proposed.Year(), proposed.Month(), proposed.Day(), 10, 0, 0, 0, proposed.Location())

        leadName := ""
        if beUUID != nil {
            if be, err := eventRepo.GetByID(ctx, *beUUID); err == nil {
                if name, ok := extractLeadName(be); ok {
                    leadName = name
                }
            }
        }
        if leadName == "" {
            leadName = "tudo bem"
        }

        message := fmt.Sprintf(
            "Sem problema, %s! Posso remarcar para %s às 10h? Me confirma se funciona.",
            leadName, proposed.Format("02/01"),
        )
        return enqueueOutboundReply(ctx, p, companyID, leadID, beUUID, message, "reschedule_proposal_v1")
    }

    // handleObjectionIntent: empathetic AI response (or static fallback).
    func handleObjectionIntent(
        ctx context.Context,
        p WhatsAppReplyPayload,
        companyID, leadID uuid.UUID,
        beUUID *uuid.UUID,
    ) error {
        templates := services.NewTemplateService()
        // Use a generic placeholder when we lack lead name.
        message := templates.RenderObjectionResponse(ctx, p.ReplyText, "tudo bem")
        return enqueueOutboundReply(ctx, p, companyID, leadID, beUUID, message, "objection_response_v1")
    }

    // handleHandoffIntent: flag the business event and emit a Redis event so
    // operators are alerted in real time. No outbound automated reply is sent.
    func handleHandoffIntent(
        ctx context.Context,
        p WhatsAppReplyPayload,
        companyID, leadID uuid.UUID,
        beUUID *uuid.UUID,
        eventRepo interface {
            GetByID(ctx context.Context, id uuid.UUID) (*domain.BusinessEvent, error)
            Update(ctx context.Context, e *domain.BusinessEvent) error
        },
    ) error {
        l := logger.FromContext(ctx)
        if beUUID != nil {
            if be, err := eventRepo.GetByID(ctx, *beUUID); err == nil {
                be.Status = "HANDOFF"
                _ = eventRepo.Update(ctx, be)
            }
        }
        publisher := sse.NewRedisPublisher()
        payload := events.NewEventPayloadV1(
            events.TypeHandoffTriggered,
            companyID.String(), leadID.String(),
            "", time.Now().Format(time.RFC3339),
        )
        payload.Metadata["reason"] = "intent_handoff"
        payload.Metadata["reply_text"] = p.ReplyText
        if beUUID != nil {
            payload.Metadata["event_id"] = beUUID.String()
        }
        if err := publisher.PublishNotification(ctx, companyID.String(), payload); err != nil {
            l.Warn("handoff_publish_failed", zap.Error(err))
        }
        return nil
    }

    // enqueueOutboundReply records the outbound turn, allocates a delivery row,
    // and enqueues the WhatsAppSend task immediately.
    func enqueueOutboundReply(
        ctx context.Context,
        p WhatsAppReplyPayload,
        companyID, leadID uuid.UUID,
        beUUID *uuid.UUID,
        message, templateID string,
    ) error {
        convRepo := repositories.NewConversationRepository(database.DB)
        deliveryRepo := repositories.NewWhatsAppDeliveryRepository(database.DB)
        deliveryID := uuid.New()

        // Outbound conversation turn (intent column unused for OUTBOUND).
        _ = convRepo.Create(ctx, &domain.ConversationTurn{
            CompanyID:  companyID,
            LeadID:     leadID,
            Direction:  events.DirectionOutbound,
            Content:    message,
            EventID:    beUUID,
            DeliveryID: &deliveryID,
            TraceID:    p.TraceID,
        })

        _ = deliveryRepo.Create(ctx, &domain.WhatsAppDelivery{
            ID:              deliveryID,
            CompanyID:       companyID,
            LeadID:          leadID,
            BusinessEventID: beUUID,
            InstanceID:      p.InstanceID,
            Phone:           p.Phone,
            TemplateID:      templateID,
            MessageText:     message,
            Status:          events.DeliveryPending,
            TraceID:         p.TraceID,
        })

        sendPayload := WhatsAppSendPayload{
            DeliveryID:  deliveryID.String(),
            CompanyID:   companyID.String(),
            LeadID:      leadID.String(),
            InstanceID:  p.InstanceID,
            Phone:       p.Phone,
            TemplateID:  templateID,
            MessageText: message,
            TraceID:     p.TraceID,
        }
        if beUUID != nil {
            sendPayload.BusinessEventID = beUUID.String()
        }
        task, opts, err := NewWhatsAppSendTask(sendPayload, time.Time{})
        if err != nil {
            return err
        }
        client := GetAsynqClient()
        defer client.Close()
        _, err = client.EnqueueContext(ctx, task, opts...)
        return err
    }

    // extractLeadName tries to read a lead_name from the business event payload.
    func extractLeadName(be *domain.BusinessEvent) (string, bool) {
        var p struct {
            LeadName string `json:"lead_name"`
        }
        if err := json.Unmarshal(be.Payload, &p); err != nil {
            return "", false
        }
        return p.LeadName, p.LeadName != ""
    }
    ```

    ```go
    package handlers

    import (
        "fmt"
        "net/http"
        "strings"

        "github.com/digitalcombo/sherlock-scraper/backend/internal/queue"
        "github.com/gofiber/fiber/v2"

    // WhatsAppWebhookHandler exposes the InternalAuth-gated endpoint that
    // WhatsMiau (or any internal forwarder) calls when a WhatsApp reply
    // arrives. The handler does ONLY validation + enqueue + 202 to satisfy
    // the strict response-time budget of upstream webhook senders.
    type WhatsAppWebhookHandler struct{}

    func NewWhatsAppWebhookHandler() *WhatsAppWebhookHandler {
        return &WhatsAppWebhookHandler{}
    }

    type whatsappReplyRequest struct {
        CompanyID       string `json:"company_id"`
        LeadID          string `json:"lead_id"`
        BusinessEventID string `json:"business_event_id,omitempty"`
        InstanceID      string `json:"instance_id"`
        Phone           string `json:"phone"`
        ReplyText       string `json:"reply_text"`
        TraceID         string `json:"trace_id,omitempty"`
    }

    // Reply enqueues TaskWhatsAppReply and returns 202.
    // Response-time budget: <50ms p95. Do not call DB or LLM here.
    func (h *WhatsAppWebhookHandler) Reply(c *fiber.Ctx) error {
        var req whatsappReplyRequest
        if err := c.BodyParser(&req); err != nil {
            return c.Status(http.StatusBadRequest).JSON(fiber.Map{"error": "invalid request body"})
        }

        // Minimal field validation.
        missing := []string{}
        if req.CompanyID == "" {
            missing = append(missing, "company_id")
        }
        if req.LeadID == "" {
            missing = append(missing, "lead_id")
        }
        if req.InstanceID == "" {
            missing = append(missing, "instance_id")
        }
        if req.Phone == "" {
            missing = append(missing, "phone")
        }
        if strings.TrimSpace(req.ReplyText) == "" {
            missing = append(missing, "reply_text")
        }
        if len(missing) > 0 {
            return c.Status(http.StatusBadRequest).JSON(fiber.Map{
                "error":   "missing fields",
                "missing": missing,
            })
        }

        if len(req.ReplyText) > 4000 {
            return c.Status(http.StatusRequestEntityTooLarge).JSON(fiber.Map{
                "error": "reply_text too long",
            })
        }

        payload := queue.WhatsAppReplyPayload{
            CompanyID:       req.CompanyID,
            LeadID:          req.LeadID,
            BusinessEventID: req.BusinessEventID,
            InstanceID:      req.InstanceID,
            Phone:           req.Phone,
            ReplyText:       req.ReplyText,
            TraceID:         req.TraceID,
        }
        task, opts, err := queue.NewWhatsAppReplyTask(payload)
        if err != nil {
            return c.Status(http.StatusInternalServerError).JSON(fiber.Map{
                "error": fmt.Sprintf("build task: %v", err),
            })
        }
        client := queue.GetAsynqClient()
        defer client.Close()
        info, err := client.EnqueueContext(c.Context(), task, opts...)
        if err != nil {
            zap.L().Error("whatsapp_reply_enqueue_failed", zap.Error(err))
            return c.Status(http.StatusInternalServerError).JSON(fiber.Map{
                "error": "enqueue failed",
            })
        }

        return c.Status(http.StatusAccepted).JSON(fiber.Map{
            "task_id": info.ID,
            "queued":  true,
        })
    }
    ```
