    ```go
    func ParseIntent(raw string) IntentType {
        switch normalizeIntent(raw) {
        case "CONFIRM":
            return IntentConfirm
        case "RESCHEDULE":
            return IntentReschedule
        case "OBJECTION":
            return IntentObjection
        case "HANDOFF":
            return IntentHandoff

        }

        out := make([]byte, 0, len(s))
        for i := 0; i < len(s); i++ {
            c := s[i]
            if c >= 'a' && c <= 'z' {
                c -= 32
            }
            if c >= 'A' && c <= 'Z' {
                out = append(out, c)
            }
        }
        return string(out)
    ```

    ```go
    package domain

    import (
        "time"

        "github.com/digitalcombo/sherlock-scraper/backend/pkg/events"
        "github.com/google/uuid"
        "gorm.io/gorm"
    )

    // ConversationTurn is one durable record of a single inbound or outbound
    // WhatsApp message in a lead's conversation history. The Redis context
    // window (conv:context:{lead_id}) is the fast cache; this table is the
    // source of truth for compliance, handoff context display, and re-hydration
    // when Redis evicts.
    //
    // Multi-tenant isolation: every query MUST filter by company_id (V4 ASVS).
    type ConversationTurn struct {
        ID         uuid.UUID                    `gorm:"type:uuid;primaryKey" json:"id"`
        CompanyID  uuid.UUID                    `gorm:"type:uuid;index:idx_conv_company_created,priority:1;not null" json:"company_id"`
        LeadID     uuid.UUID                    `gorm:"type:uuid;index:idx_conv_lead_created,priority:1;not null" json:"lead_id"`
        Direction  events.ConversationDirection `gorm:"type:varchar(10);not null" json:"direction"`
        Content    string                       `gorm:"type:text;not null" json:"content"`
        Intent     string                       `gorm:"type:varchar(20)" json:"intent,omitempty"`
        EventID    *uuid.UUID                   `gorm:"type:uuid;index" json:"event_id,omitempty"`     // optional FK to business_events
        DeliveryID *uuid.UUID                   `gorm:"type:uuid;index" json:"delivery_id,omitempty"`  // optional FK to whatsapp_deliveries
        TraceID    string                       `gorm:"type:varchar(64);index" json:"trace_id,omitempty"`
        CreatedAt  time.Time                    `gorm:"index:idx_conv_lead_created,priority:2,sort:desc;index:idx_conv_company_created,priority:2,sort:desc" json:"created_at"`
        UpdatedAt  time.Time                    `json:"updated_at"`
        DeletedAt  gorm.DeletedAt               `gorm:"index" json:"-"`
    }

    func (c *ConversationTurn) BeforeCreate(tx *gorm.DB) error {
        if c.ID == uuid.Nil {
            c.ID = uuid.New()
        }
        return nil
    }
    ```
